The NoSQL database that
lives inside your JVM.

EmbedJNoSQL is a document and key-value database — plus Redis-style structures, wide-column families, and experimental vector search — in a single pure-Java JAR. No Docker. No daemon. No infrastructure.

Application.java 
// One line. Zero infrastructure.
try (var db = EmbedJNoSQL.inMemory()) {

    // ── Documents with nested field queries ─────────────────────
    db.documentCollection("users")
      .insert(Document.of("name", "Alice").add("city", "Nairobi"));

    List<Document> users = db.documentCollection("users")
      .find(Query.eq("city", "Nairobi").and(Query.gte("age", 18)));

    // ── Redis-style structures ───────────────────────────────────
    db.listBucket("job-queue").rpush("tasks", "send-email", "resize-image");
    db.hashBucket("profiles").hset("u1", "tier", "GOLD");

    // ── Key-Value with TTL ───────────────────────────────────────
    db.keyValueBucket("sessions").put("tok-abc", "user-1");
    db.keyValueBucket("sessions").expire("tok-abc", Duration.ofMinutes(30));
}
